cmake_minimum_required(VERSION 3.24)
project(AstralSigil VERSION 2.0.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
include(FetchContent)
set(SPDLOG_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPDLOG_INSTALL ON CACHE BOOL "" FORCE)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_XAUDIO_WIN8 OFF CACHE BOOL "" FORCE)
set(BUILD_XAUDIO_WIN10 OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
FetchContent_Declare(spdlog GIT_REPOSITORY https://github.com/gabime/spdlog.git GIT_TAG 6fa36017cfd5731d617e1a934f0e5ea9c4445b13 GIT_SHALLOW FALSE)
FetchContent_Declare(directxtk GIT_REPOSITORY https://github.com/microsoft/DirectXTK.git GIT_TAG 861f3e7df437b50d1f28c964769329e9b23a50fb GIT_SHALLOW FALSE)
FetchContent_Declare(rapidcsv GIT_REPOSITORY https://github.com/d99kris/rapidcsv.git GIT_TAG 3c8131af7596329c50380ef3f8975e45742b1d60 GIT_SHALLOW FALSE)
FetchContent_Declare(commonlib GIT_REPOSITORY https://github.com/alandtse/CommonLibSSE-NG.git GIT_TAG cce63b4c8aacd2766c21f44b408df92a3ad9e95c GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(spdlog directxtk rapidcsv)
add_library(Microsoft::DirectXTK ALIAS DirectXTK)
foreach(package IN ITEMS spdlog directxtk)
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/packages/${package}")
    file(WRITE "${CMAKE_BINARY_DIR}/packages/${package}/${package}Config.cmake" "set(${package}_FOUND TRUE)\n")
    set(${package}_DIR "${CMAKE_BINARY_DIR}/packages/${package}" CACHE PATH "" FORCE)
endforeach()
set(RAPIDCSV_INCLUDE_DIRS "${rapidcsv_SOURCE_DIR}/src" CACHE PATH "" FORCE)
FetchContent_MakeAvailable(commonlib)
target_compile_options(CommonLibSSE PRIVATE /MP6)
add_library(AstralSigilPortal SHARED src/main.cpp)
target_compile_definitions(AstralSigilPortal PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(AstralSigilPortal PRIVATE /utf-8 /W4)
target_link_libraries(AstralSigilPortal PRIVATE CommonLibSSE::CommonLibSSE)
enable_testing()
foreach(name IN ITEMS portal casting runtime)
    add_executable(test_${name} tests/${name}.cpp)
    target_include_directories(test_${name} PRIVATE src)
    if(name STREQUAL "casting")
        target_link_libraries(test_${name} PRIVATE CommonLibSSE::CommonLibSSE)
    endif()
    add_test(NAME ${name} COMMAND test_${name})
endforeach()
